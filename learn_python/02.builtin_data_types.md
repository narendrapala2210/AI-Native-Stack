# 2. Built-in Data Types & Data Structures

### 2.1 Numeric Types: int, float, complex, decimal, fractions

```python
from decimal import Decimal, getcontext
from fractions import Fraction

# 1. Arbitrary-precision integer
big_num = 10 ** 35
print("Arbitrary precision int:", big_num)

# Binary, Octal, Hexadecimal literals
b_val = 0b1010       # 10
o_val = 0o755        # 493
h_val = 0xFF         # 255

# 2. Floating-point representation issue & Decimal solution:
print(0.1 + 0.2)         # 0.30000000000000004 (Binary floating point representation)
print(0.1 + 0.2 == 0.3)  # False!

# Solution: Decimal module for exact arithmetic (finance / science)
getcontext().prec = 6
d1 = Decimal("0.1")
d2 = Decimal("0.2")
print("Decimal result:", d1 + d2)        # Decimal('0.3')
print(d1 + d2 == Decimal("0.3"))         # True

# 3. Fractions module
f1 = Fraction(1, 3)
f2 = Fraction(1, 6)
print("Fraction addition:", f1 + f2)     # 1/2

# 4. Complex numbers
z1 = 3 + 4j
z2 = complex(2, -1)
print("Complex multiplication:", z1 * z2)    # (10+5j)
print("Absolute / Magnitude:", abs(z1))      # 5.0
```

### 2.2 Booleans & Truth Value Testing
`bool` is a subclass of `int`. `True == 1` and `False == 0`.

```python
print(isinstance(True, int))  # True
print(True + True + 10)       # 12

# Falsy values: None, False, 0, 0.0, 0j, Decimal(0), "", (), [], {}, set(), range(0)
# All other values are Truthy.

# Short-circuit logic:
# 'and' returns the first falsy operand, or the last operand if all are truthy
print([] and "Python")       # []
print("Python" and [1, 2])   # [1, 2]

# 'or' returns the first truthy operand, or the last operand if all are falsy
print("" or "Default")       # "Default"
print("First" or "Second")   # "First"
```

### 2.3 Strings: Unicode, Slicing, Methods & Modern Formatting

```python
text = "Python Programming Masterclass"

# 1. Advanced Slicing [start:stop:step]
print(text[0:6])         # "Python"
print(text[:6])          # "Python"
print(text[7:])          # "Programming Masterclass"
print(text[::2])         # "Pto rgamn atrla"
print(text[::-1])        # "ssalcretsaM gnimmargorP nohtyP" (Reverse string)

# 2. String Methods
clean = "   AI Full Stack Engineer   ".strip()
words = clean.split(" ")
rejoined = "-".join(words)
print("Rejoined string:", rejoined)
print(clean.startswith("AI"))
print(clean.replace("Engineer", "Architect"))

# 3. String Formatting Evolution
name = "Alice"
score = 98.4567

# Form 1: C-style % (legacy)
print("Student: %s, Score: %.2f" % (name, score))

# Form 2: str.format()
print("Student: {0}, Score: {1:.2f}".format(name, score))

# Form 3: f-strings (Python 3.6+, fastest and modern)
print(f"Student: {name.upper()}, Score: {score:.2f}")

# f-strings self-documenting expression (Python 3.8+)
x = 42
print(f"{x=}")  # Outputs: x=42

# 4. Raw and Byte Strings
raw_path = r"C:\new_folder\data.txt"  # Escape characters \n, \t are ignored
byte_data = b"Hello\x20World"         # Raw byte sequence
print("Decoded bytes:", byte_data.decode("utf-8"))
```

### 2.4 Lists: Dynamic Arrays, Operations, Shallow vs Deep Copy

```python
import copy

items = [10, 20, 30, 40]

items.append(50)          # O(1) amortized
items.insert(1, 15)       # O(n) shifts elements
items.extend([60, 70])    # In-place extension
popped = items.pop()      # O(1) removes last element
items.remove(15)          # O(n) removes first matching value
items.sort(reverse=True)  # In-place Timsort O(n log n)
print("Processed items:", items)

# Shallow vs Deep Copy:
original = [[1, 2, 3], ["a", "b", "c"]]
shallow = list(original)       # or original.copy()
deep = copy.deepcopy(original)

original[0].append(999)

print("Original:", original)  # [[1, 2, 3, 999], ['a', 'b', 'c']]
print("Shallow:", shallow)    # [[1, 2, 3, 999], ['a', 'b', 'c']] <-- Mutated!
print("Deep:", deep)          # [[1, 2, 3], ['a', 'b', 'c']]       <-- Safe!
```

### 2.5 Tuples: Immutability, Unpacking & Namedtuples

```python
from collections import namedtuple

# Single element tuple syntax requires trailing comma
not_a_tup = ("single")   # type: str
is_a_tup = ("single",)   # type: tuple

# Extended Unpacking (PEP 3132)
data = (10, 20, 30, 40, 50)
first, *middle, last = data
print(f"First: {first}, Middle: {middle}, Last: {last}")
# First: 10, Middle: [20, 30, 40], Last: 50

# Variable swapping
x, y = 100, 200
x, y = y, x
print(f"x={x}, y={y}")

# namedtuple: struct-like readable immutable records
Point = namedtuple("Point", ["x", "y", "label"])
p = Point(x=10.5, y=20.8, label="Origin")
print(p.x, p.y, p.label)
print(p[0], p[1])
```

### 2.6 Dictionaries: Hash Tables, Keys, and collections Specialized Dicts

Dictionaries use hash tables. Keys must be **hashable** (immutable and implementing `__hash__` and `__eq__`).

```python
from collections import defaultdict, Counter, OrderedDict, ChainMap

user = {"name": "Charlie", "role": "Engineer", "level": 4}

# Safe key access and default assignment
print(user.get("department", "Engineering"))
user.setdefault("skills", []).append("Python")

# collections.defaultdict (avoids explicit key checking)
grouped = defaultdict(list)
records = [("dev", "Alice"), ("qa", "Bob"), ("dev", "Charlie")]
for team, person in records:
    grouped[team].append(person)
print("Grouped defaultdict:", dict(grouped))

# collections.Counter (frequency tallying)
votes = ["apple", "banana", "apple", "cherry", "apple", "banana"]
counter = Counter(votes)
print("Counter results:", counter.most_common(2))  # [('apple', 3), ('banana', 2)]

# collections.ChainMap (search multiple dicts sequentially)
env_config = {"DEBUG": True}
default_config = {"DEBUG": False, "PORT": 8000}
config = ChainMap(env_config, default_config)
print("Resolved DEBUG:", config["DEBUG"])  # True
print("Resolved PORT:", config["PORT"])    # 8000
```

### 2.7 Sets & Frozensets: Mathematical Operations & Efficiency

Sets provide **$O(1)$ average time complexity** membership lookups.

```python
a = {1, 2, 3, 4, 5}
b = {4, 5, 6, 7, 8}

print("Union (|):", a | b)                 # {1, 2, 3, 4, 5, 6, 7, 8}
print("Intersection (&):", a & b)          # {4, 5}
print("Difference (-):", a - b)            # {1, 2, 3}
print("Symmetric Diff (^):", a ^ b)        # {1, 2, 3, 6, 7, 8}
print("Subset check:", {1, 2}.issubset(a)) # True

# frozenset: Immutable set, can be used as dictionary keys
frozen = frozenset([1, 2, 3])
cache_dict = {frozen: "Precomputed value"}
print(cache_dict[frozen])
```

### 2.8 NoneType, Ellipsis (`...`), and Sentinels

```python
# None comparison
val = None
print(val is None)  # Always use 'is', never '=='

# Sentinel pattern
_MISSING = object()

def lookup(key, store, default=_MISSING):
    if key in store:
        return store[key]
    if default is not _MISSING:
        return default
    raise KeyError(f"Key {key} not found")

test_store = {"data": None}
print("Found None value safely:", lookup("data", test_store))
```

### 2.9 Enumerations with the `enum` Module (`Enum`, `IntEnum`, `Flag`, `auto`)

Enumerations provide type-safe, human-readable symbolic constants.

```python
from enum import Enum, IntEnum, Flag, auto, unique

# 1. Standard Enum with @unique decorator to prevent duplicate values
@unique
class OrderStatus(Enum):
    PENDING = "pending"
    PROCESSING = "processing"
    SHIPPED = "shipped"
    DELIVERED = "delivered"
    CANCELLED = "cancelled"

status = OrderStatus.SHIPPED
print("Enum name:", status.name)    # "SHIPPED"
print("Enum value:", status.value)  # "shipped"
print(status is OrderStatus.SHIPPED) # True

# 2. auto() assigns sequential values automatically
class Priority(IntEnum):
    LOW = auto()      # 1
    MEDIUM = auto()   # 2
    HIGH = auto()     # 3
    CRITICAL = auto() # 4

print("IntEnum comparison:", Priority.HIGH > Priority.LOW) # True

# 3. Flag and bitwise combination (Permissions)
class Permission(Flag):
    READ = auto()     # 1 (0b001)
    WRITE = auto()    # 2 (0b010)
    EXECUTE = auto()  # 4 (0b100)

user_permissions = Permission.READ | Permission.WRITE
print("Has READ?", Permission.READ in user_permissions)       # True
print("Has EXECUTE?", Permission.EXECUTE in user_permissions) # False
```

### 2.10 Advanced Text & Unicode: `unicodedata`, `str.translate`, `maketrans`

```python
import unicodedata
import string

# 1. Unicode Normalization (NFC, NFD, NFKC, NFKD)
# Two strings can look identical to humans but have completely different byte representations!
s1 = "café"            # 'e' with acute accent (single code point U+00E9)
s2 = "cafe\u0301"      # 'e' followed by combining acute accent (U+0301)

print("Direct equality:", s1 == s2) # False!
print("NFC Normalized:", unicodedata.normalize("NFC", s1) == unicodedata.normalize("NFC", s2)) # True

# 2. str.translate() and str.maketrans() (Fastest text transformation)
# Strip all punctuation and translate vowels to uppercase in C speed
vowels_to_upper = str.maketrans("aeiou", "AEIOU", string.punctuation)
dirty_text = "hello, world! python is amazing."
cleaned = dirty_text.translate(vowels_to_upper)
print("Translated text:", cleaned)  # "hEllO wOrld pythOn Is AmAzIng"
```

### Example 1: Financial Currency Ledger with Exact Precision
Using `float` in banking or billing systems leads to rounding errors. This example builds an accurate transaction ledger using `decimal.Decimal` and `fractions.Fraction`.

```python
from decimal import Decimal, ROUND_HALF_UP
from fractions import Fraction

class FinancialLedger:
    def __init__(self, tax_rate: str = "0.0825"):
        self.tax_rate = Decimal(tax_rate)
        self.entries = []

    def add_transaction(self, description: str, unit_price: str, quantity: int, split_fraction: tuple = (1, 1)):
        # Calculate split ratio exactly using Fraction
        ratio = Fraction(*split_fraction)
        unit = Decimal(unit_price)
        subtotal = (unit * quantity * Decimal(ratio.numerator)) / Decimal(ratio.denominator)
        tax = (subtotal * self.tax_rate).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
        total = subtotal + tax
        self.entries.append({"desc": description, "subtotal": subtotal, "tax": tax, "total": total})

    def print_summary(self):
        grand_total = sum(e["total"] for e in self.entries)
        for e in self.entries:
            print(f"Item: {e['desc']:<15} Subtotal: ${e['subtotal']:.2f} Tax: ${e['tax']:.2f} Total: ${e['total']:.2f}")
        print(f"Grand Total: ${grand_total:.2f}")

ledger = FinancialLedger()
ledger.add_transaction("Cloud Hosting", "149.99", 1, (1, 1))
ledger.add_transaction("Shared DB Unit", "80.00", 1, (1, 2))  # 50% split
ledger.print_summary()
```

### Example 2: High-Performance Log Sanitizer with Unicode Normalization
Clean and scrub incoming user-submitted text by normalizing unicode characters and translating away sensitive symbols at C speed.

```python
import unicodedata
import string

class LogSanitizer:
    def __init__(self):
        # Strip control characters, punctuation, and map digits to '#'
        self.translation_table = str.maketrans(
            string.punctuation + string.digits,
            " " * len(string.punctuation) + "#" * len(string.digits)
        )

    def sanitize(self, raw_input: str) -> str:
        # 1. Normalize unicode (convert accents, symbols to canonical forms)
        normalized = unicodedata.normalize("NFKD", raw_input)
        # 2. Encode to ASCII bytes, ignoring non-ASCII artifacts, then decode
        ascii_clean = normalized.encode("ascii", "ignore").decode("utf-8")
        # 3. Translate symbols and mask numbers
        scrubbed = ascii_clean.translate(self.translation_table)
        # 4. Collapse extra whitespace
        return " ".join(scrubbed.split())

sanitizer = LogSanitizer()
dirty_log = "Alert! Usér: Alex-9921 failed login at IP: [192.168.1.105] (status: 404)!"
print("Sanitized Output:", sanitizer.sanitize(dirty_log))
```

### Example 3: Role-Based Access Control (RBAC) with `enum.Flag`
Combine discrete user permissions into single integer bitmasks that can be stored compactly and evaluated in sub-nanosecond speeds.

```python
from enum import Flag, auto

class UserPermission(Flag):
    NONE = 0
    VIEW_REPORTS = auto()     # 1
    EDIT_REPORTS = auto()     # 2
    DELETE_REPORTS = auto()   # 4
    MANAGE_USERS = auto()     # 8
    BILLING_ACCESS = auto()   # 16
    
    # Composite presets
    ANALYST = VIEW_REPORTS
    MANAGER = VIEW_REPORTS | EDIT_REPORTS | DELETE_REPORTS
    SUPERADMIN = VIEW_REPORTS | EDIT_REPORTS | DELETE_REPORTS | MANAGE_USERS | BILLING_ACCESS

def authorize_action(user_role: UserPermission, required_perm: UserPermission) -> bool:
    return required_perm in user_role

analyst_user = UserPermission.ANALYST
manager_user = UserPermission.MANAGER

print("Can analyst view reports?", authorize_action(analyst_user, UserPermission.VIEW_REPORTS))   # True
print("Can analyst delete reports?", authorize_action(analyst_user, UserPermission.DELETE_REPORTS)) # False
print("Can manager delete reports?", authorize_action(manager_user, UserPermission.DELETE_REPORTS)) # True
```
