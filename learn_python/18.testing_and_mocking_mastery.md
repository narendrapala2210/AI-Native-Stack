# 18. Testing & Mocking Mastery

### 18.1 Unit Testing with `unittest`

```python
import unittest

def divide_positive(a: float, b: float) -> float:
    if b <= 0:
        raise ValueError("Divisor must be strictly positive.")
    return a / b

class TestMathDivision(unittest.TestCase):
    def setUp(self):
        # Runs BEFORE each test
        self.base_numerator = 100.0

    def tearDown(self):
        # Runs AFTER each test
        pass

    def test_successful_division(self):
        result = divide_positive(self.base_numerator, 4.0)
        self.assertEqual(result, 25.0)

    def test_negative_or_zero_divisor_raises_error(self):
        with self.assertRaises(ValueError) as context:
            divide_positive(self.base_numerator, 0.0)
        self.assertIn("Divisor must be strictly positive", str(context.exception))

if __name__ == "__main__":
    unittest.main(argv=["first-arg-is-ignored"], exit=False)
```

### 18.2 Mocking, Spying & Patching with `unittest.mock`

`unittest.mock` allows isolating code under test from external dependencies (network, databases, third-party APIs).

```python
from unittest.mock import Mock, MagicMock, patch

# 1. Basic Mock and assertions
mock_mailer = Mock()
mock_mailer.send_email.return_value = {"status": "DELIVERED"}

response = mock_mailer.send_email(recipient="alice@test.io", subject="Hello")
print("Mock response:", response)
mock_mailer.send_email.assert_called_once_with(recipient="alice@test.io", subject="Hello")

# 2. Side effects (Dynamic returns or exceptions)
mock_api = Mock()
mock_api.get_data.side_effect = [10, 20, Exception("Server Unavailable")]
print("Call 1:", mock_api.get_data())  # 10
print("Call 2:", mock_api.get_data())  # 20
try:
    mock_api.get_data()  # Raises Exception
except Exception as e:
    print("Caught mock side effect error:", e)

# 3. Patching external libraries using context manager
class WeatherService:
    @staticmethod
    def get_temperature_online(city: str) -> float:
        import urllib.request
        # Imagine a real blocking HTTP call here
        return 72.5

with patch.object(WeatherService, "get_temperature_online", return_value=85.0) as patched_method:
    val = WeatherService.get_temperature_online("San Francisco")
    print(f"Patched temperature: {val}°F (actual HTTP call never happened!)")
    patched_method.assert_called_once_with("San Francisco")
```

### 18.3 Doctests: Executable Documentation in Docstrings

Doctests ensure that code examples in your documentation remain up to date and never become stale.

```python
import doctest

def factorial(n: int) -> int:
    """
    Computes the factorial of a non-negative integer.

    >>> factorial(0)
    1
    >>> factorial(1)
    1
    >>> factorial(5)
    120
    >>> factorial(-1)
    Traceback (most recent call last):
      ...
    ValueError: n must be non-negative
    """
    if n < 0:
        raise ValueError("n must be non-negative")
    if n <= 1:
        return 1
    return n * factorial(n - 1)

# Run doctests automatically
results = doctest.testmod()
print(f"Doctest results: {results.attempted} tests run, {results.failed} failed.")
```

### 18.4 Modern Testing with `pytest` (Fixtures & Parametrize)

`pytest` is the most popular Python testing framework.

```python
# Save as test_sample.py and run via 'pytest'
"""
import pytest

# Fixture provides reusable test dependencies
@pytest.fixture
def sample_user_dict():
    return {"id": 1, "username": "alex", "tier": "gold"}

def test_user_tier(sample_user_dict):
    assert sample_user_dict["tier"] == "gold"

# Data-driven parameterized testing
@pytest.mark.parametrize("input_num, expected", [
    (2, True),
    (3, True),
    (4, False),
    (15, False),
    (17, True)
])
def test_is_prime(input_num, expected):
    from math import isqrt
    def is_prime(n):
        return n > 1 and all(n % i != 0 for i in range(2, isqrt(n) + 1))
    assert is_prime(input_num) == expected
"""
```

## 🛠️ Practical Hands-On & Real-World Examples

### Example 1: Mocking Third-Party HTTP APIs with `@patch` and `MagicMock`
Test service logic in total isolation without hitting external network endpoints.

```python
from unittest.mock import patch, MagicMock
import unittest

class PaymentGatewayClient:
    def charge_card(self, token: str, amount: float) -> dict:
        import urllib.request
        # Simulated external HTTP request
        return {"status": "SUCCESS", "id": "tx_9988"}

def process_order(client: PaymentGatewayClient, token: str, amount: float) -> bool:
    res = client.charge_card(token, amount)
    return res.get("status") == "SUCCESS"

class TestOrderPayment(unittest.TestCase):
    @patch.object(PaymentGatewayClient, "charge_card")
    def test_successful_charge(self, mock_charge):
        mock_charge.return_value = {"status": "SUCCESS", "id": "tx_fake_001"}
        
        client = PaymentGatewayClient()
        success = process_order(client, "tok_visa", 49.99)
        
        self.assertTrue(success)
        mock_charge.assert_called_once_with("tok_visa", 49.99)
        print("Mock test passed successfully!")

suite = unittest.TestLoader().loadTestsFromTestCase(TestOrderPayment)
unittest.TextTestRunner(verbosity=2).run(suite)
```

### Example 2: Data-Driven Parametrized Input Testing (pytest pattern)
Test multiple edge cases and inputs against expected outputs without duplicate test functions.

```python
def validate_password_strength(password: str) -> bool:
    # Requires at least 8 chars, 1 digit, 1 uppercase
    if len(password) < 8:
        return False
    if not any(c.isdigit() for c in password):
        return False
    if not any(c.isupper() for c in password):
        return False
    return True

# Parametrized test runner
test_cases = [
    ("short", False),
    ("NoDigitsHere", False),
    ("alllowercase123", False),
    ("ValidPass123", True),
    ("SuperSecure2026!", True)
]

for pwd, expected in test_cases:
    result = validate_password_strength(pwd)
    assert result == expected, f"Failed on '{pwd}': expected {expected}, got {result}"
    print(f"PASS: '{pwd}' -> {result}")
```

### Example 3: Self-Testing String Normalizer with `doctest`
Embed verifiable, executable test cases directly inside function docstrings.

```python
import doctest

def sanitize_slug(text: str) -> str:
    r"""
    Convert raw titles into URL-safe slugs.

    >>> sanitize_slug("Hello World!")
    'hello-world'
    >>> sanitize_slug("  Python 3.12 Deep Dive  ")
    'python-312-deep-dive'
    >>> sanitize_slug("AI & Machine Learning")
    'ai-machine-learning'
    """
    clean = "".join(c.lower() if c.isalnum() or c.isspace() else "" for c in text)
    return "-".join(clean.split())

res = doctest.testmod()
print(f"Doctest finished: {res.attempted} tests run, {res.failed} failures.")
```
