### 5.3 The Mutable Default Parameter Trap & The Fix

```python
# The Trap:
def flawed_logger(msg, logs=[]):
    logs.append(msg)
    return logs

print(flawed_logger("First"))  # ['First']
print(flawed_logger("Second")) # ['First', 'Second'] (List is shared!)

# The Clean Pythonic Solution:
def correct_logger(msg, logs=None):
    if logs is None:
        logs = []
    logs.append(msg)
    return logs

print(correct_logger("First"))  # ['First']
print(correct_logger("Second")) # ['Second']
```

### 5.4 Variable Scope: LEGB Rule, `global` and `nonlocal`

```python
system_state = "IDLE"

def manager():
    local_state = "READY"

    def worker():
        nonlocal local_state  # Modify enclosing variable
        global system_state   # Modify module-level variable

        local_state = "BUSY"
        system_state = "RUNNING"

    worker()
    print("Enclosing local_state:", local_state)

manager()
print("Global system_state:", system_state)
```

### 5.5 First-Class Functions, Lambdas, map, filter, reduce

```python
from functools import reduce

nums = [1, 2, 3, 4, 5, 6]

# Anonymous lambda function
square = lambda x: x ** 2

# map
squared = list(map(lambda x: x ** 2, nums))

# filter
evens = list(filter(lambda x: x % 2 == 0, nums))

# reduce
product = reduce(lambda acc, x: acc * x, nums, 1)

print(f"{squared=}")
print(f"{evens=}")
print(f"{product=}")
```

### 5.6 Closures & Free Variables

```python
def power_generator(exponent: int):
    # 'exponent' is a free variable captured in closure
    def power(base: int) -> int:
        return base ** exponent
    return power

square_fn = power_generator(2)
cube_fn = power_generator(3)

print("Square of 4:", square_fn(4))  # 16
print("Cube of 4:", cube_fn(4))      # 64
print("Captured exponent:", square_fn.__closure__[0].cell_contents)  # 2
```

### 5.7 Recursion, Base Cases & Recursion Limits

```python
import sys

def sum_digits(n: int) -> int:
    if n < 10:  # Base case
        return n
    return (n % 10) + sum_digits(n // 10)  # Recursive step

print("Sum of digits in 9874:", sum_digits(9874))  # 28
print("Python recursion limit:", sys.getrecursionlimit())
```

## 🛠️ Practical Hands-On & Real-World Examples

### Example 1: Sliding-Window Rate Limiter via Closures
Build an API rate limiter function factory that maintains state in a closure without global variables or external classes.

```python
import time

def create_rate_limiter(max_requests: int, window_seconds: float):
    timestamps = []

    def is_allowed() -> bool:
        nonlocal timestamps
        now = time.time()
        # Discard expired timestamps outside window
        timestamps = [t for t in timestamps if now - t < window_seconds]
        if len(timestamps) < max_requests:
            timestamps.append(now)
            return True
        return False

    return is_allowed

limiter = create_rate_limiter(max_requests=2, window_seconds=0.1)

print("Call 1 allowed?", limiter())  # True
print("Call 2 allowed?", limiter())  # True
print("Call 3 allowed?", limiter())  # False (Limit reached!)
time.sleep(0.12)
print("Call 4 allowed after delay?", limiter())  # True
```

### Example 2: Functional Pipeline Composition Operator
Compose multiple transformation functions into a single pipeline where data flows through each step (`f(g(h(x)))`).

```python
from functools import reduce

def compose(*functions):
    # Composes functions right-to-left: compose(f, g)(x) == f(g(x))
    return reduce(lambda f, g: lambda x: f(g(x)), functions, lambda x: x)

def pipe(*functions):
    # Composes functions left-to-right: pipe(f, g)(x) == g(f(x))
    return reduce(lambda f, g: lambda x: g(f(x)), functions, lambda x: x)

strip_whitespace = lambda s: s.strip()
remove_dashes = lambda s: s.replace("-", "")
to_uppercase = lambda s: s.upper()

clean_serial_number = pipe(strip_whitespace, remove_dashes, to_uppercase)
raw_input = "   sn-889-ab-01   "
print("Cleaned serial:", clean_serial_number(raw_input))  # "SN889AB01"
```

### Example 3: Nested Directory Tree Walker with Recursion Guard
Recursively walk hierarchical data structures (nested dictionary folders) with safety limits against deep stack overflow.

```python
file_system = {
    "root": {
        "src": {
            "app.py": "code",
            "utils": {
                "helpers.py": "code"
            }
        },
        "README.md": "text"
    }
}

def walk_tree(node: dict, current_depth: int = 0, max_depth: int = 5):
    if current_depth > max_depth:
        print("  " * current_depth + "[!] Max recursion depth exceeded!")
        return

    for name, child in node.items():
        indent = "  " * current_depth
        if isinstance(child, dict):
            print(f"{indent}📁 {name}/")
            walk_tree(child, current_depth + 1, max_depth)
        else:
            print(f"{indent}📄 {name} ({child})")

walk_tree(file_system)
```
