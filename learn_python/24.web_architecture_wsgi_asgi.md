# 24. Web Architecture: WSGI, ASGI & Server Internals

### 24.1 WSGI Protocol (PEP 3333): Building a Minimal WSGI Application

**WSGI (Web Server Gateway Interface)** is the synchronous standard connecting web servers (Gunicorn, uWSGI) to web frameworks (Flask, Django).

```python
from wsgiref.simple_server import make_server

# The WSGI callable interface: takes 'environ' dict and 'start_response' callable
def simple_wsgi_app(environ, start_response):
    status = "200 OK"
    headers = [("Content-Type", "application/json; charset=utf-8")]
    start_response(status, headers)

    path = environ.get("PATH_INFO", "/")
    method = environ.get("REQUEST_METHOD", "GET")

    response_body = f'{{"path": "{path}", "method": "{method}", "server": "WSGI"}}'
    return [response_body.encode("utf-8")]

# Testing WSGI app in-process
environ_sample = {"PATH_INFO": "/api/v1/health", "REQUEST_METHOD": "GET"}
def mock_start_response(status, headers):
    print(f"[WSGI] Status: {status}, Headers: {headers}")

response = simple_wsgi_app(environ_sample, mock_start_response)
print("WSGI Response Body:", b"".join(response).decode("utf-8"))
```

### 24.2 ASGI Protocol: Building a Minimal Asynchronous Web Server

**ASGI (Asynchronous Server Gateway Interface)** is the modern standard powering async frameworks (FastAPI, Starlette) and servers (Uvicorn).

```python
import json

# The ASGI callable: async def app(scope, receive, send)
async def minimal_asgi_app(scope, receive, send):
    assert scope["type"] == "http"

    # 1. Send HTTP Response Start (headers)
    await send({
        "type": "http.response.start",
        "status": 200,
        "headers": [
            [b"content-type", b"application/json"],
        ]
    })

    # 2. Send HTTP Response Body
    payload = json.dumps({"status": "healthy", "protocol": "ASGI 3.0"}).encode("utf-8")
    await send({
        "type": "http.response.body",
        "body": payload,
    })

# In production: uvicorn main:minimal_asgi_app --port 8000
```

### 24.3 Middleware Pipelines in WSGI and ASGI

Middleware wraps an application to intercept requests and responses (authentication, CORS, rate limiting, logging).

```python
# WSGI Middleware Example
class LoggingMiddleware:
    def __init__(self, app):
        self.app = app

    def __call__(self, environ, start_response):
        print(f"[LOG MIDDLEWARE] Inbound request to: {environ.get('PATH_INFO')}")
        return self.app(environ, start_response)

wrapped_app = LoggingMiddleware(simple_wsgi_app)
wrapped_app(environ_sample, mock_start_response)
```

## 🛠️ Practical Hands-On & Real-World Examples

### Example 1: Custom Minimal Routing Micro-Framework with WSGI
Build a functioning HTTP router and dispatcher from scratch following PEP 3333.

```python
import json

class MiniRouterWSGI:
    def __init__(self):
        self.routes = {}

    def route(self, path: str):
        def decorator(handler):
            self.routes[path] = handler
            return handler
        return decorator

    def __call__(self, environ, start_response):
        path = environ.get("PATH_INFO", "/")
        handler = self.routes.get(path)

        if handler:
            status = "200 OK"
            headers = [("Content-Type", "application/json")]
            body = json.dumps(handler(environ)).encode("utf-8")
        else:
            status = "404 Not Found"
            headers = [("Content-Type", "text/plain")]
            body = b"Route not found."

        start_response(status, headers)
        return [body]

app = MiniRouterWSGI()

@app.route("/api/ping")
def ping_handler(environ):
    return {"message": "pong", "status": "UP"}

# Simulate server call
def mock_start_response(status, headers):
    print("Response Status:", status)

response = app({"PATH_INFO": "/api/ping"}, mock_start_response)
print("Response Body:", b"".join(response).decode())
```

### Example 2: Asynchronous Server-Sent Events (SSE) Streaming with ASGI
Build an ASGI endpoint that streams events asynchronously to connected clients.

```python
import asyncio

async def sse_event_stream_app(scope, receive, send):
    assert scope["type"] == "http"
    
    # 1. Send SSE Headers
    await send({
        "type": "http.response.start",
        "status": 200,
        "headers": [
            [b"content-type", b"text/event-stream"],
            [b"cache-control", b"no-cache"],
        ]
    })
    
    # 2. Stream 3 real-time ticks
    for i in range(1, 4):
        msg = f"data: tick #{i}\n\n".encode("utf-8")
        await send({
            "type": "http.response.body",
            "body": msg,
            "more_body": True if i < 3 else False
        })
        await asyncio.sleep(0.01)
```

### Example 3: Request Timing & Authentication Middleware Pipeline
Build a composable middleware pipeline that records latency and validates Authorization headers.

```python
class TimingAndAuthMiddleware:
    def __init__(self, app, valid_token: str = "Bearer secret123"):
        self.app = app
        self.valid_token = valid_token

    def __call__(self, environ, start_response):
        import time
        start_time = time.perf_counter()
        
        auth_header = environ.get("HTTP_AUTHORIZATION", "")
        if auth_header != self.valid_token:
            start_response("401 Unauthorized", [("Content-Type", "text/plain")])
            return [b"Missing or invalid Bearer token."]

        def custom_start_response(status, headers, exc_info=None):
            duration = (time.perf_counter() - start_time) * 1000
            headers.append(("X-Response-Time-Ms", f"{duration:.2f}"))
            return start_response(status, headers, exc_info)

        return self.app(environ, custom_start_response)

wrapped = TimingAndAuthMiddleware(app)
res_unauth = wrapped({}, lambda s, h, e=None: print("Unauth test:", s))
print("Unauth output:", b"".join(res_unauth).decode())
```
