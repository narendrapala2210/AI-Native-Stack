# 13. Modern Type Annotations & Pydantic Validation

### 13.1 Type Hints: Primitives, Generics (`list[T]`), Unions (`|`), Optional

Python 3.5+ introduced type hints (PEP 484), with continuous modernization (PEP 585, PEP 604):
- Built-in generics: `list[str]`, `dict[str, int]`, `set[float]`, `tuple[int, ...]`
- Union syntax with pipe: `int | str` (replaces `Union[int, str]`)
- Optional syntax: `str | None` (replaces `Optional[str]`)

```python
def process_user_records(
    user_ids: list[int],
    metadata: dict[str, str | int],
    fallback: str | None = None
) -> tuple[int, bool]:
    total_processed = len(user_ids)
    is_valid = bool(metadata)
    return total_processed, is_valid

result: tuple[int, bool] = process_user_records([1, 2, 3], {"role": "admin", "level": 5})
print(f"Processed: {result[0]}, Status: {result[1]}")
```

### 13.2 Advanced Typing: `Callable`, `Literal`, `TypedDict`, `Protocol`

```python
from typing import Callable, Literal, TypedDict, Protocol

# 1. Callable: Callable[[ArgTypes...], ReturnType]
Transformer = Callable[[str], str]

def apply_formatting(text: str, formatter: Transformer) -> str:
    return formatter(text)

# 2. Literal: Restrict values to specific exact constants
ThemeMode = Literal["dark", "light", "high-contrast"]

def set_theme(theme: ThemeMode) -> None:
    print("Setting theme to:", theme)

# 3. TypedDict: Dictionary with a strictly specified key-value schema
class UserProfile(TypedDict):
    username: str
    age: int
    is_active: bool

user: UserProfile = {"username": "alex", "age": 29, "is_active": True}

# 4. Protocol: Structural subtyping (Static Duck Typing, PEP 544)
class Renderable(Protocol):
    def render(self) -> str:
        ...

class MarkdownDocument:
    def render(self) -> str:
        return "# Markdown Title"

class HTMLDocument:
    def render(self) -> str:
        return "<h1>HTML Title</h1>"

def display(doc: Renderable) -> None:
    # Any object with a render() -> str method satisfies Renderable without inheriting it!
    print("Rendered:", doc.render())

display(MarkdownDocument())
display(HTMLDocument())
```

### 13.3 Generic Classes & Functions with `TypeVar` (PEP 695 syntax)

In Python 3.12+, generics have first-class syntax via the `[T]` parameter syntax (PEP 695):

```python
# Modern Python 3.12+ syntax:
class Stack[T]:
    def __init__(self) -> None:
        self._items: list[T] = []

    def push(self, item: T) -> None:
        self._items.append(item)

    def pop(self) -> T:
        return self._items.pop()

int_stack = Stack[int]()
int_stack.push(10)
int_stack.push(20)
print("Popped from stack:", int_stack.pop())

# Generic Function:
def get_first_element[T](elements: list[T]) -> T | None:
    return elements[0] if elements else None

print("First item:", get_first_element(["alpha", "beta"]))
```

### 13.4 Pydantic v2: Models, Field Validation & Serialization

Pydantic is the industry standard for runtime data validation, parsing, and serialization in modern Python (FastAPI, LangChain, LlamaIndex).

```python
from pydantic import BaseModel, Field, EmailStr, field_validator, model_validator
from datetime import datetime

class UserModel(BaseModel):
    id: int
    username: str = Field(min_length=3, max_length=20)
    email: str
    is_premium: bool = False
    created_at: datetime = Field(default_factory=datetime.now)

    @field_validator("username")
    @classmethod
    def username_must_be_alphanumeric(cls, v: str) -> str:
        if not v.isalnum():
            raise ValueError("Username must be alphanumeric.")
        return v.lower()

# Runtime validation from raw dictionary:
raw_input = {
    "id": 101,
    "username": "SuperDev99",
    "email": "dev@company.io",
    "is_premium": True
}

validated_user = UserModel.model_validate(raw_input)
print(f"Validated Model: {validated_user.username} (ID: {validated_user.id})")

# Serialization to dict or JSON:
user_dict = validated_user.model_dump()
user_json = validated_user.model_dump_json(indent=2)
print("Dumped JSON:\n", user_json)
```

### Example 1: Microservice Request/Response Schema with Pydantic v2
Validate incoming JSON payloads with nested models, field constraints, and custom validators.

```python
from pydantic import BaseModel, Field, field_validator
from typing import List

class OrderItem(BaseModel):
    sku: str = Field(min_length=3, max_length=12)
    quantity: int = Field(gt=0, le=100)
    price: float = Field(gt=0.0)

class OrderCheckoutPayload(BaseModel):
    customer_id: int
    items: List[OrderItem] = Field(min_length=1)

    @field_validator("items")
    @classmethod
    def validate_unique_skus(cls, items: List[OrderItem]):
        skus = [i.sku for i in items]
        if len(skus) != len(set(skus)):
            raise ValueError("Duplicate SKUs detected in order items!")
        return items

    @property
    def total_cost(self) -> float:
        return sum(item.quantity * item.price for item in self.items)

payload = {
    "customer_id": 1002,
    "items": [
        {"sku": "SKU-A01", "quantity": 2, "price": 29.99},
        {"sku": "SKU-B02", "quantity": 1, "price": 49.99}
    ]
}

order = OrderCheckoutPayload.model_validate(payload)
print(f"Order valid! Customer: {order.customer_id}, Total: ${order.total_cost:.2f}")
```

### Example 2: Structural Subtyping Plugin System with `typing.Protocol`
Define static interfaces without forcing subclasses to inherit from a shared base class.

```python
from typing import Protocol

class MessageNotifier(Protocol):
    def send_notification(self, recipient: str, body: str) -> bool:
        ...

# Unrelated classes that satisfy the Protocol implicitly:
class SlackService:
    def send_notification(self, recipient: str, body: str) -> bool:
        print(f"[SLACK to #{recipient}]: {body}")
        return True

class DiscordService:
    def send_notification(self, recipient: str, body: str) -> bool:
        print(f"[DISCORD to @{recipient}]: {body}")
        return True

def notify_user(notifier: MessageNotifier, user: str, msg: str):
    notifier.send_notification(user, msg)

notify_user(SlackService(), "alerts", "Build 402 succeeded")
notify_user(DiscordService(), "dev-chat", "Deployment complete")
```

### Example 3: Generic In-Memory Thread-Safe Cache with `Generic[T]`
Build a generic typed cache class using Python 3.12+ type parameter syntax.

```python
from threading import Lock

class GenericCache[T]:
    def __init__(self):
        self._store: dict[str, T] = {}
        self._lock = Lock()

    def set(self, key: str, value: T) -> None:
        with self._lock:
            self._store[key] = value

    def get(self, key: str) -> T | None:
        with self._lock:
            return self._store.get(key)

# Type-safe integer cache
int_cache = GenericCache[int]()
int_cache.set("count", 42)
print("Int cache value:", int_cache.get("count"))

# Type-safe string cache
str_cache = GenericCache[str]()
str_cache.set("status", "HEALTHY")
print("Str cache value:", str_cache.get("status"))
```
