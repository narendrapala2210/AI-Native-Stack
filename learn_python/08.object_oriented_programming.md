# 8. Object-Oriented Programming (OOP) Deep Dive

### 8.1 Classes, Instances, `__init__`, and `self`

```python
class BankAccount:
    """Represents a commercial bank account."""

    def __init__(self, account_holder: str, initial_balance: float = 0.0):
        # 'self' refers to the concrete instance being initialized
        self.account_holder = account_holder
        self.balance = initial_balance

    def deposit(self, amount: float) -> float:
        if amount <= 0:
            raise ValueError("Deposit amount must be positive.")
        self.balance += amount
        return self.balance

account = BankAccount("John Doe", 1000.0)
print(f"Holder: {account.account_holder}, Balance: ${account.deposit(250.0)}")
```

### 8.2 Class Attributes vs Instance Attributes & `__dict__`

- **Class attributes** are shared across **all** instances of the class.
- **Instance attributes** are stored in the instance's unique `__dict__`.

```python
class Employee:
    # Class attribute (shared by all instances)
    company_name = "Tech Innovations Corp"
    minimum_wage = 15.0

    def __init__(self, name: str, hourly_rate: float):
        # Instance attributes (unique to this object)
        self.name = name
        self.hourly_rate = hourly_rate

emp1 = Employee("Alice", 45.0)
emp2 = Employee("Bob", 50.0)

print(emp1.company_name)  # "Tech Innovations Corp"
print(emp2.company_name)  # "Tech Innovations Corp"

# Modifying class attribute via class updates for all:
Employee.company_name = "Global AI Systems"
print(emp1.company_name)  # "Global AI Systems"

# Inspecting instance namespace vs class namespace:
print("emp1 __dict__:", emp1.__dict__)
print("Employee class has minimum_wage:", "minimum_wage" in Employee.__dict__)
```

### 8.3 Instance Methods, `@classmethod`, and `@staticmethod`

```python
import json

class User:
    species = "Homo sapiens"

    def __init__(self, username: str, email: str):
        self.username = username
        self.email = email

    # 1. Instance method: operates on instance state ('self')
    def get_profile(self) -> str:
        return f"{self.username} <{self.email}>"

    # 2. Class method: operates on class state ('cls'), excellent for alternative constructors
    @classmethod
    def from_json(cls, json_string: str):
        data = json.loads(json_string)
        return cls(username=data["username"], email=data["email"])

    # 3. Static method: isolated utility function; doesn't take 'self' or 'cls'
    @staticmethod
    def is_valid_email(email: str) -> bool:
        return "@" in email and "." in email

# Alternative constructor call
u = User.from_json('{"username": "dev_alex", "email": "alex@dev.internal"}')
print(u.get_profile())
print("Email valid:", User.is_valid_email(u.email))
```

### 8.4 Encapsulation, Name Mangling & `@property`

Python doesn't have strict `private` keywords, but follows conventions:
- `_protected`: Convention warning that attribute is internal.
- `__private`: Name mangling modifies the attribute name to `_ClassName__attribute` to prevent accidental subclass overrides.
- `@property`: Clean getters, setters, and deleters.

```python
class ThermalSensor:
    def __init__(self, celsius: float = 0.0):
        self._celsius = celsius
        self.__secret_device_id = "DEV-9981"  # Name mangled

    @property
    def celsius(self) -> float:
        """Getter for celsius temperature."""
        return self._celsius

    @celsius.setter
    def celsius(self, value: float):
        """Setter with validation."""
        if value < -273.15:
            raise ValueError("Temperature below absolute zero is impossible!")
        self._celsius = value

    @property
    def fahrenheit(self) -> float:
        """Read-only derived property."""
        return (self._celsius * 9 / 5) + 32

sensor = ThermalSensor(25.0)
print(f"Celsius: {sensor.celsius}°C, Fahrenheit: {sensor.fahrenheit}°F")

sensor.celsius = 30.0  # Invokes setter
print(f"Updated Fahrenheit: {sensor.fahrenheit}°F")

# Demonstrating name mangling:
# print(sensor.__secret_device_id) -> Raises AttributeError
print("Mangled access:", sensor._ThermalSensor__secret_device_id)
```

### 8.5 Inheritance, `super()`, Diamond Problem & MRO (C3 Linearization)

Python supports **Multiple Inheritance**. When methods clash, Python determines the execution hierarchy using the **Method Resolution Order (MRO)** computed by the C3 Linearization algorithm.

```python
class Base:
    def execute(self):
        print("Base.execute")

class ServiceA(Base):
    def execute(self):
        print("ServiceA.execute start")
        super().execute()
        print("ServiceA.execute end")

class ServiceB(Base):
    def execute(self):
        print("ServiceB.execute start")
        super().execute()
        print("ServiceB.execute end")

class HybridPipeline(ServiceA, ServiceB):
    def execute(self):
        print("HybridPipeline.execute start")
        super().execute()
        print("HybridPipeline.execute end")

pipeline = HybridPipeline()
pipeline.execute()

# Inspecting the MRO
print("\nMRO Resolution Order:")
for idx, cls in enumerate(HybridPipeline.mro()):
    print(f"{idx}: {cls.__name__}")
```

### 8.6 Polymorphism and Duck Typing

> *"If it walks like a duck and quacks like a duck, it's a duck."*

In Python, polymorphism does not depend on explicit interface inheritance.

```python
class PostgresStorage:
    def save(self, data: str):
        print(f"Saved '{data}' to PostgreSQL database table.")

class S3Storage:
    def save(self, data: str):
        print(f"Uploaded '{data}' to Amazon S3 bucket.")

class LocalDiskStorage:
    def save(self, data: str):
        print(f"Wrote '{data}' to local NVMe drive.")

def persist_record(storage_backend, record: str):
    # Duck typing: we only care that storage_backend has a callable 'save' method!
    storage_backend.save(record)

for backend in [PostgresStorage(), S3Storage(), LocalDiskStorage()]:
    persist_record(backend, "User Activity Log #4092")
```

### 8.7 Abstract Base Classes (`abc.ABC`, `@abstractmethod`)

Abstract Base Classes define formal contracts that subclasses **must** implement, preventing instantiation of incomplete classes.

```python
from abc import ABC, abstractmethod

class PaymentGateway(ABC):
    @abstractmethod
    def authenticate(self, api_key: str) -> bool:
        """Authenticate with the payment processor."""
        pass

    @abstractmethod
    def process_payment(self, amount: float) -> str:
        """Charge the given amount and return transaction ID."""
        pass

class StripeGateway(PaymentGateway):
    def authenticate(self, api_key: str) -> bool:
        return api_key.startswith("sk_live_")

    def process_payment(self, amount: float) -> str:
        return f"STRIPE-TX-{int(amount * 100)}"

# gateway = PaymentGateway() # Raises TypeError: Can't instantiate abstract class
stripe = StripeGateway()
print("Authenticated:", stripe.authenticate("sk_live_98745"))
print("Transaction:", stripe.process_payment(99.50))
```

### 8.8 Complete Dunder (Magic) Methods Matrix

Dunder (`__double_underscore__`) methods allow user-defined classes to hook into Python's core language features, operators, and protocols.

```python
class Vector2D:
    # 1. Lifecycle
    def __init__(self, x: float, y: float):
        self.x = x
        self.y = y

    # 2. String representation: __repr__ (unambiguous/developer) vs __str__ (user-friendly)
    def __repr__(self):
        return f"Vector2D({self.x}, {self.y})"

    def __str__(self):
        return f"({self.x}i + {self.y}j)"

    # 3. Arithmetic operators (+, -, *)
    def __add__(self, other):
        if not isinstance(other, Vector2D):
            return NotImplemented
        return Vector2D(self.x + other.x, self.y + other.y)

    def __mul__(self, scalar: float):
        return Vector2D(self.x * scalar, self.y * scalar)

    # Right-hand arithmetic for scalar * vector
    def __rmul__(self, scalar: float):
        return self.__mul__(scalar)

    # 4. Equality and Comparison
    def __eq__(self, other):
        return isinstance(other, Vector2D) and self.x == other.x and self.y == other.y

    # 5. Length / Magnitude
    def __abs__(self):
        return (self.x ** 2 + self.y ** 2) ** 0.5

    # 6. Container subscription [0], [1]
    def __getitem__(self, index: int):
        if index == 0:
            return self.x
        elif index == 1:
            return self.y
        raise IndexError("Vector2D index out of range (0 or 1 only)")

    # 7. Callable instance
    def __call__(self):
        return f"Vector magnitude is {abs(self):.2f}"

v1 = Vector2D(3, 4)
v2 = Vector2D(1, 2)

print("Str representation:", str(v1))
print("Repr representation:", repr(v1))
print("Vector addition (v1 + v2):", v1 + v2)
print("Scalar multiplication (3 * v1):", 3 * v1)
print("Magnitude abs(v1):", abs(v1))
print("Indexing v1[0]:", v1[0])
print("Calling instance v1():", v1())
```

### 8.9 Memory Optimization with `__slots__`

By default, Python stores instance attributes in a dynamic dictionary (`__dict__`). Defining `__slots__` bypasses `__dict__` and creates fixed-size C array allocations, saving significant memory when creating millions of instances.

```python
import sys

# Standard class with __dict__
class RegularPoint:
    def __init__(self, x, y):
        self.x = x
        self.y = y

# Optimized class with __slots__
class SlottedPoint:
    __slots__ = ("x", "y")  # Restricts attributes to x and y only; no __dict__!
    def __init__(self, x, y):
        self.x = x
        self.y = y

rp = RegularPoint(10, 20)
sp = SlottedPoint(10, 20)

print("Regular point has __dict__:", hasattr(rp, "__dict__"))
print("Slotted point has __dict__:", hasattr(sp, "__dict__"))
print("Regular point size (dict):", sys.getsizeof(rp.__dict__))
# Note: sp.z = 30 -> Raises AttributeError (Prevents accidental dynamic attributes)
```

### 8.10 Modern Dataclasses (`@dataclass`)
Introduced in Python 3.7 (PEP 557), `@dataclass` auto-generates `__init__`, `__repr__`, `__eq__`, hashing, and comparison operators based on type annotations.

```python
from dataclasses import dataclass, field
from typing import List

@dataclass(order=True, frozen=False, slots=True)
class Product:
    # Sort order will compare sort_index first
    sort_index: float = field(init=False, repr=False)
    name: str
    price: float
    tags: List[str] = field(default_factory=list)

    def __post_init__(self):
        # Calculated after __init__ completes
        self.sort_index = self.price

p1 = Product("Mechanical Keyboard", 120.0, ["hardware", "peripherals"])
p2 = Product("Ergonomic Mouse", 60.0, ["hardware"])

print(p1)
print("p2 < p1 (ordered by price):", p2 < p1)  # True (60.0 < 120.0)
```

### 8.11 Customizing Object Copy Semantics: `__copy__` and `__deepcopy__`

You can control exactly how objects are duplicated by the standard `copy` module by implementing `__copy__()` and `__deepcopy__(memo)`.

```python
import copy

class GraphNode:
    def __init__(self, name: str, connections: list = None):
        self.name = name
        self.connections = connections if connections is not None else []

    def __copy__(self):
        # Shallow copy: new node, but connection list is referenced
        return GraphNode(self.name, self.connections)

    def __deepcopy__(self, memo):
        # Deep copy: prevent infinite loops in cyclic graphs using 'memo'
        if id(self) in memo:
            return memo[id(self)]
        
        cloned_node = GraphNode(self.name)
        memo[id(self)] = cloned_node
        cloned_node.connections = [copy.deepcopy(neighbor, memo) for neighbor in self.connections]
        return cloned_node

# Testing cyclic graph copying
node_a = GraphNode("Node A")
node_b = GraphNode("Node B")
node_a.connections.append(node_b)
node_b.connections.append(node_a)  # Cycle!

deep_cloned_a = copy.deepcopy(node_a)
print("Original and clone are separate instances:", node_a is not deep_cloned_a)
print("Clone cycle resolved cleanly:", deep_cloned_a.connections[0].connections[0] is deep_cloned_a)
```

### Example 1: Multi-Currency Money Class with Full Operator Overloading
A production-grade Money class that safely prevents adding distinct currencies without conversion and supports comparison, hashing, and arithmetic.

```python
from decimal import Decimal

class Money:
    __slots__ = ("_amount", "_currency")

    def __init__(self, amount: str | float | Decimal, currency: str = "USD"):
        self._amount = Decimal(str(amount)).quantize(Decimal("0.01"))
        self._currency = currency.upper()

    @property
    def amount(self) -> Decimal:
        return self._amount

    @property
    def currency(self) -> str:
        return self._currency

    def __repr__(self):
        return f"Money('{self._amount}', '{self._currency}')"

    def __str__(self):
        return f"{self._amount} {self._currency}"

    def __add__(self, other):
        if not isinstance(other, Money) or self._currency != other._currency:
            raise ValueError(f"Cannot add {self._currency} and {getattr(other, 'currency', type(other))}")
        return Money(self._amount + other._amount, self._currency)

    def __mul__(self, factor: float | int):
        return Money(self._amount * Decimal(str(factor)), self._currency)

    def __eq__(self, other):
        return isinstance(other, Money) and self._amount == other._amount and self._currency == other._currency

    def __lt__(self, other):
        if not isinstance(other, Money) or self._currency != other._currency:
            raise ValueError("Cannot compare distinct currencies")
        return self._amount < other._amount

price1 = Money("49.95", "USD")
price2 = Money("15.00", "USD")
total = (price1 + price2) * 2

print("Total price:", total)
print("price1 > price2:", price1 > price2)
```

### Example 2: Enterprise Entity Repository using Abstract Base Classes
Enforce clean architecture database contracts using `abc.ABC` and `@abstractmethod`.

```python
from abc import ABC, abstractmethod
from typing import Optional, List

class UserRepository(ABC):
    @abstractmethod
    def get_by_id(self, user_id: int) -> Optional[dict]:
        # Fetch user record by ID.
        pass

    @abstractmethod
    def save(self, user: dict) -> None:
        # Persist or update user record.
        pass

class InMemoryUserRepository(UserRepository):
    def __init__(self):
        self._db = {}

    def get_by_id(self, user_id: int) -> Optional[dict]:
        return self._db.get(user_id)

    def save(self, user: dict) -> None:
        self._db[user["id"]] = user
        print(f"Saved user: {user['username']}")

repo = InMemoryUserRepository()
repo.save({"id": 101, "username": "claire"})
print("Fetched:", repo.get_by_id(101))
```

### Example 3: Memory-Optimized Point Cloud with `__slots__` and `@dataclass`
Store millions of 3D spatial points with minimal RAM consumption using `@dataclass(slots=True, frozen=True)`.

```python
from dataclasses import dataclass
import sys

@dataclass(slots=True, frozen=True)
class SpatialPoint:
    x: float
    y: float
    z: float

    def distance_from_origin(self) -> float:
        return (self.x**2 + self.y**2 + self.z**2) ** 0.5

p = SpatialPoint(1.5, 2.0, 3.5)
print(f"Point: {p}, Distance: {p.distance_from_origin():.2f}")
print("Point object memory footprint:", sys.getsizeof(p), "bytes")
```
