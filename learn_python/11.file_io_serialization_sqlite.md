### 11.1 File Modes, Encodings, Buffering & Context Management

| Mode | Description |
| :--- | :--- |
| `'r'` | Read (default). Fails if file does not exist. |
| `'w'` | Write. Truncates/overwrites file or creates new file. |
| `'a'` | Append. Writes to the end without truncating. |
| `'x'` | Exclusive creation. Fails if file already exists. |
| `'b'` | Binary mode (e.g., `'rb'`, `'wb'`). Returns `bytes`. |
| `'+'` | Updating (read and write, e.g., `'r+'`, `'w+'`). |

```python
# Safe writing and reading with context manager & explicit encoding
with open("sample_output.txt", "w", encoding="utf-8") as f:
    f.write("Line 1: UTF-8 encoded text: 🚀\n")
    f.writelines(["Line 2: Next item\n", "Line 3: Final item\n"])

with open("sample_output.txt", "r", encoding="utf-8") as f:
    for line in f:  # Memory-efficient line-by-line streaming
        print("Read line:", line.rstrip())
```

### 11.2 Modern Filesystem Navigation with `pathlib.Path`

`pathlib` provides an object-oriented, cross-platform approach to path manipulation (replaces legacy `os.path`).

```python
from pathlib import Path

# Path building with the '/' operator
base_dir = Path("my_project") / "data" / "processed"
print("Constructed path:", base_dir)

# Inspection methods
sample_file = Path("sample_output.txt")
print("File exists?", sample_file.exists())
print("Is file?", sample_file.is_file())
print("Suffix / Extension:", sample_file.suffix)
print("Filename stem:", sample_file.stem)
print("Absolute path:", sample_file.resolve())

# Direct reading/writing shortcuts without explicit open():
sample_file.write_text("Quick write with pathlib", encoding="utf-8")
print("Read text:", sample_file.read_text(encoding="utf-8"))

# Cleanup
if sample_file.exists():
    sample_file.unlink()
```

```python
import json

payload = {

csv_file = Path("users.csv")

    {"id": 1, "username": "alice", "role": "admin"},
    {"id": 2, "username": "bob", "role": "member"}
]

with open(csv_file, "w", newline="", encoding="utf-8") as f:
    fieldnames = ["id", "username", "role"]
    writer = csv.DictWriter(f, fieldnames=fieldnames)
    writer.writeheader()

    reader = csv.DictReader(f)
    for row in reader:
        print(f"User {row['id']}: {row['username']} ({row['role']})")

if csv_file.exists():
    csv_file.unlink()
```

### 11.4 Binary Serialization with `pickle` & Security Implications

`pickle` serializes arbitrary Python object hierarchies into byte streams.

> [!CAUTION]
> **Never unpickle data received from an untrusted source!** Malicious pickles can execute arbitrary shell commands during deserialization. Use JSON, Protocol Buffers, or MessagePack for untrusted network transfer.

```python
import pickle

data_structure = {"key": "secret", "values": (1, 2, [3, 4])}

# Serialize
pickled_bytes = pickle.dumps(data_structure)
print("Pickled byte stream:", pickled_bytes[:30], "...")

# Deserialize
restored = pickle.loads(pickled_bytes)
print("Restored identical structure:", restored == data_structure)
```

### 11.5 Embedded Database with `sqlite3`

```python
import sqlite3

# Connect to in-memory database
with sqlite3.connect(":memory:") as conn:
    # Use Row factory to access columns by name like dicts
    conn.row_factory = sqlite3.Row
    cursor = conn.cursor()

    # Create table
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS agents (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            capability TEXT NOT NULL
        )
    """)

    # Parameterized queries (prevents SQL Injection!)
    agents = [
        ("Antigravity", "Autonomous Software Engineering"),
        ("DataBot", "Analytics and Insights")
    ]
    cursor.executemany("INSERT INTO agents (name, capability) VALUES (?, ?)", agents)
    conn.commit()

    # Querying
    cursor.execute("SELECT * FROM agents WHERE name = ?", ("Antigravity",))
    row = cursor.fetchone()
    print(f"Agent: {row['name']} | Capability: {row['capability']}")
```

### 11.6 High-Level File Operations: `shutil`, `tempfile`, and `zipfile`

```python
import shutil
import tempfile
import zipfile
from pathlib import Path

# 1. tempfile: Safe temporary files and directories that clean up automatically
with tempfile.TemporaryDirectory() as tmpdir:
    temp_dir_path = Path(tmpdir)
    doc_path = temp_dir_path / "notes.txt"
    doc_path.write_text("Confidential scratchpad content", encoding="utf-8")
    print(f"Temporary file created at: {doc_path} (exists: {doc_path.exists()})")
# Temporary directory and its contents are automatically deleted here!
print("Temp directory cleaned up:", not temp_dir_path.exists())

# 2. shutil: High-level filesystem manipulations (disk usage, recursive copying)
usage = shutil.disk_usage(".")
print(f"Free disk space: {usage.free // (1024 ** 3)} GB")

# 3. zipfile: Creating and inspecting compressed ZIP archives in memory or disk
archive_path = Path("demo_archive.zip")
with zipfile.ZipFile(archive_path, mode="w", compression=zipfile.ZIP_DEFLATED) as zf:
    zf.writestr("archive_notes.txt", "Compressed file stored in archive.")

# Inspecting archive contents without unzipping everything
with zipfile.ZipFile(archive_path, mode="r") as zf:
    print("Archive file list:", zf.namelist())
    print("Read from zip directly:", zf.read("archive_notes.txt").decode("utf-8"))

if archive_path.exists():
    archive_path.unlink()
```

## 🛠️ Practical Hands-On & Real-World Examples

### Example 1: Atomic JSON Configuration Store with Backup & Restore
Read, validate, write, and maintain automatic backup snapshots of system configurations.

```python
import json
import shutil
from pathlib import Path

class ConfigStore:
    def __init__(self, file_path: str):
        self.file_path = Path(file_path)
        self.backup_path = self.file_path.with_suffix(".bak")

    def save(self, data: dict):
        if self.file_path.exists():
            shutil.copy2(self.file_path, self.backup_path)  # Backup existing
        self.file_path.write_text(json.dumps(data, indent=2), encoding="utf-8")
        print(f"Saved configuration to {self.file_path}")

    def load(self) -> dict:
        try:
            return json.loads(self.file_path.read_text(encoding="utf-8"))
        except (FileNotFoundError, json.JSONDecodeError):
            if self.backup_path.exists():
                print("[RECOVERY] Primary config corrupted; restoring from backup...")
                return json.loads(self.backup_path.read_text(encoding="utf-8"))
            return {}

cfg = ConfigStore("app_config.json")
cfg.save({"env": "production", "workers": 4})
print("Loaded config:", cfg.load())

# Cleanup
cfg.file_path.unlink(missing_ok=True)
cfg.backup_path.unlink(missing_ok=True)
```

### Example 2: Compressed Archive Backup System with `zipfile` and `pathlib`
Recursively gather files matching specific glob patterns and pack them into a timestamped compressed ZIP archive.

```python
import zipfile
from pathlib import Path
import tempfile

def create_project_archive(source_dir: Path, output_zip: Path, file_extension: str = "*.py"):
    with zipfile.ZipFile(output_zip, mode="w", compression=zipfile.ZIP_DEFLATED) as archive:
        for file in source_dir.rglob(file_extension):
            archive_name = file.relative_to(source_dir)
            archive.write(file, arcname=archive_name)
            print(f"Archived: {archive_name}")

with tempfile.TemporaryDirectory() as tmpdir:
    src = Path(tmpdir) / "project"
    src.mkdir()
    (src / "main.py").write_text("print('hello')", encoding="utf-8")
    (src / "utils.py").write_text("def add(a,b): return a+b", encoding="utf-8")
    
    zip_out = Path(tmpdir) / "backup.zip"
    create_project_archive(src, zip_out)
    
    with zipfile.ZipFile(zip_out, "r") as zf:
        print("Archive verified contents:", zf.namelist())
```

### Example 3: Embedded SQLite Key-Value Storage with Expiration TTL
Build a local, zero-dependency embedded key-value cache using SQLite's in-memory / local database with timestamp expiration.

```python
import sqlite3
import time

class SQLiteCache:
    def __init__(self, db_path: str = ":memory:"):
        self.conn = sqlite3.connect(db_path)
        self.conn.execute("CREATE TABLE IF NOT EXISTS cache (key TEXT PRIMARY KEY, value TEXT, expires_at REAL)")
        self.conn.commit()

    def set(self, key: str, value: str, ttl_seconds: float = 60.0):
        expires_at = time.time() + ttl_seconds
        self.conn.execute("INSERT OR REPLACE INTO cache VALUES (?, ?, ?)", (key, value, expires_at))
        self.conn.commit()

    def get(self, key: str) -> str | None:
        cur = self.conn.cursor()
        cur.execute("SELECT value, expires_at FROM cache WHERE key = ?", (key,))
        row = cur.fetchone()
        if not row:
            return None
        val, expires_at = row
        if time.time() > expires_at:
            self.conn.execute("DELETE FROM cache WHERE key = ?", (key,))
            self.conn.commit()
            return None
        return val

cache = SQLiteCache()
cache.set("session_token", "secret_9988", ttl_seconds=0.05)
print("Immediate fetch:", cache.get("session_token"))
time.sleep(0.06)
print("Fetch after TTL expiration:", cache.get("session_token"))  # None
```
