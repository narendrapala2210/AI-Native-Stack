# 17. Pythonic Design Patterns: Creational, Structural & Behavioral

### 17.1 Creational Patterns: Singleton, Factory, Builder

#### 1. The Singleton Pattern:
Ensures a class has only one instance and provides a global access point.

```python
# A. Metaclass Singleton (Most robust approach in Python)
class SingletonMeta(type):
    _instances = {}
    def __call__(cls, *args, **kwargs):
        if cls not in cls._instances:
            cls._instances[cls] = super().__call__(*args, **kwargs)
        return cls._instances[cls]

class DatabaseConnectionPool(metaclass=SingletonMeta):
    def __init__(self):
        self.connection_id = "CONN-8890"

pool1 = DatabaseConnectionPool()
pool2 = DatabaseConnectionPool()
print("Singleton pool instances are identical:", pool1 is pool2)  # True

# B. Note: In Python, modules themselves are already singletons!
# Importing 'config.py' across 20 files returns the exact same cached module object.
```

#### 2. The Factory Pattern:
Decouples client code from concrete object creation classes.

```python
from abc import ABC, abstractmethod

class NotificationSender(ABC):
    @abstractmethod
    def send(self, message: str) -> str:
        pass

class EmailNotification(NotificationSender):
    def send(self, message: str) -> str:
        return f"EMAIL SENT: {message}"

class SMSNotification(NotificationSender):
    def send(self, message: str) -> str:
        return f"SMS SENT: {message}"

class NotificationFactory:
    _registry = {
        "email": EmailNotification,
        "sms": SMSNotification
    }

    @classmethod
    def create(cls, channel: str) -> NotificationSender:
        sender_class = cls._registry.get(channel.lower())
        if not sender_class:
            raise ValueError(f"Unknown notification channel: {channel}")
        return sender_class()

client = NotificationFactory.create("email")
print(client.send("Your build has succeeded."))
```

#### 3. The Builder Pattern:
Constructs complex objects step-by-step with a fluent interface.

```python
class QueryBuilder:
    def __init__(self, table: str):
        self._table = table
        self._fields = ["*"]
        self._where_clauses = []
        self._limit_count = None

    def select(self, *fields):
        self._fields = list(fields)
        return self  # Return self for method chaining

    def where(self, condition: str):
        self._where_clauses.append(condition)
        return self

    def limit(self, count: int):
        self._limit_count = count
        return self

    def build(self) -> str:
        fields_sql = ", ".join(self._fields)
        query = f"SELECT {fields_sql} FROM {self._table}"
        if self._where_clauses:
            query += " WHERE " + " AND ".join(self._where_clauses)
        if self._limit_count is not None:
            query += f" LIMIT {self._limit_count}"
        return query

sql = (
    QueryBuilder("users")
    .select("id", "username", "email")
    .where("is_active = 1")
    .where("age >= 18")
    .limit(25)
    .build()
)
print("Constructed SQL Query:\n", sql)
```

### 17.2 Structural Patterns: Adapter, Facade, Proxy

#### 1. The Adapter Pattern:
Translates an interface for a class into a format another client expects.

```python
# Legacy 3rd party service
class LegacyXMLPaymentService:
    def process_xml_transaction(self, xml_payload: str):
        return f"<status>APPROVED</status><raw>{xml_payload}</raw>"

# Modern client expects JSON dictionary interface
class PaymentAdapter:
    def __init__(self, legacy_service: LegacyXMLPaymentService):
        self.legacy_service = legacy_service

    def pay(self, amount: float, account_id: str) -> dict:
        xml_string = f"<pay><acc>{account_id}</acc><amt>{amount}</amt></pay>"
        raw_res = self.legacy_service.process_xml_transaction(xml_string)
        return {"status": "SUCCESS", "amount": amount, "raw_response": raw_res}

adapter = PaymentAdapter(LegacyXMLPaymentService())
print("Adapter output:", adapter.pay(150.0, "ACC-459"))
```

#### 2. The Facade Pattern:
Provides a simplified high-level interface over a complex ecosystem of subsystems.

```python
class AudioDecoder:
    def decode(self, file): return f"AudioDecoded({file})"

class VideoDecoder:
    def decode(self, file): return f"VideoDecoded({file})"

class SubtitleParser:
    def parse(self, file): return f"SubtitlesParsed({file})"

# Facade encapsulates audio, video, and subtitle complexities
class MultimediaPlayerFacade:
    def __init__(self):
        self.audio = AudioDecoder()
        self.video = VideoDecoder()
        self.subtitles = SubtitleParser()

    def play_movie(self, movie_path: str):
        a = self.audio.decode(movie_path)
        v = self.video.decode(movie_path)
        s = self.subtitles.parse(movie_path)
        print(f"Streaming movie successfully: {v}, {a}, {s}")

player = MultimediaPlayerFacade()
player.play_movie("matrix_resurrections.mkv")
```

#### 3. The Proxy Pattern:
Controls access to an expensive or protected object through lazy initialization or authorization.

```python
class ExpensiveHeavyResource:
    def __init__(self):
        print(">>> Heavy resource loaded into 16GB memory! <<<")

    def query(self, param):
        return f"Processed query for '{param}'"

class LazyResourceProxy:
    def __init__(self):
        self._resource = None  # Not loaded yet!

    def query(self, param):
        if self._resource is None:
            print("[PROXY] Instantiating heavy resource on first demand...")
            self._resource = ExpensiveHeavyResource()
        return self._resource.query(param)

proxy = LazyResourceProxy()
print("Proxy initialized (no memory cost yet).")
print(proxy.query("User Request #1"))
print(proxy.query("User Request #2"))  # Already loaded; reuses instance
```

### 17.3 Behavioral Patterns: Strategy, Observer (Pub/Sub), Command

#### 1. The Strategy Pattern:
Enables selecting an algorithm's implementation dynamically at runtime.

```python
from typing import Callable

# Functional Strategy pattern: In Python, strategies can just be functions!
def no_discount(amount: float) -> float:
    return amount

def seasonal_discount(amount: float) -> float:
    return amount * 0.85

def vip_discount(amount: float) -> float:
    return amount * 0.70

class OrderContext:
    def __init__(self, amount: float, discount_strategy: Callable[[float], float] = no_discount):
        self.amount = amount
        self.strategy = discount_strategy

    def calculate_final_price(self) -> float:
        return self.strategy(self.amount)

order = OrderContext(100.0, vip_discount)
print(f"VIP final price: ${order.calculate_final_price():.2f}")
```

#### 2. The Observer Pattern (Publisher-Subscriber):
Maintains a list of dependents and notifies them automatically of state changes.

```python
class EventDispatcher:
    def __init__(self):
        self._subscribers = {}

    def subscribe(self, event_type: str, callback: Callable):
        self._subscribers.setdefault(event_type, []).append(callback)

    def publish(self, event_type: str, data: dict):
        for callback in self._subscribers.get(event_type, []):
            callback(data)

dispatcher = EventDispatcher()

# Observers
def email_logger(data):
    print(f"[EMAIL SERVICE] Alert dispatched to {data['email']}")

def audit_logger(data):
    print(f"[AUDIT LOG] Recorded event at timestamp {data['timestamp']}")

dispatcher.subscribe("user_registered", email_logger)
dispatcher.subscribe("user_registered", audit_logger)

dispatcher.publish("user_registered", {"email": "alex@ai.org", "timestamp": "2026-09-14 12:00:00"})
```

### Example 1: Thread-Safe Database Connection Pool (Singleton Pattern)
Ensure that only a single instance of a connection pool exists across multithreaded requests.

```python
import threading

class ThreadSafeSingletonMeta(type):
    _instances = {}
    _lock = threading.Lock()

    def __call__(cls, *args, **kwargs):
        with cls._lock:
            if cls not in cls._instances:
                cls._instances[cls] = super().__call__(*args, **kwargs)
        return cls._instances[cls]

class DatabasePool(metaclass=ThreadSafeSingletonMeta):
    def __init__(self):
        self.pool_id = "POOL-MAIN-001"
        self.max_connections = 10

pool_a = DatabasePool()
pool_b = DatabasePool()
print("Are both connection pools identical?", pool_a is pool_b)  # True
print(f"Pool ID: {pool_a.pool_id}")
```

### Example 2: Dynamic Payment Gateway Selector (Strategy Pattern)

```python
from typing import Callable

# Strategies as clean functions
def process_stripe(amount: float) -> str:
    return f"Paid ${amount:.2f} via Stripe card processing"

def process_crypto(amount: float) -> str:
```

### Example 3: Real-Time Event Dispatcher (Observer / Pub-Sub Pattern)
Decouple microservice events from consumer notification handlers.

```python
class EventDispatcher:
    def __init__(self):
        self._listeners = {}

    def subscribe(self, event_name: str, callback):
        self._listeners.setdefault(event_name, []).append(callback)

    def dispatch(self, event_name: str, payload: dict):
        for callback in self._listeners.get(event_name, []):
            callback(payload)

# Observers
def notify_slack(data):
    print(f"[SLACK ALERT] Deployment of {data['service']} completed!")

def update_metrics(data):
    print(f"[DATADOG METRIC] increment_counter('deploys', tags=['{data['service']}'])")

dispatcher = EventDispatcher()
dispatcher.subscribe("deploy_success", notify_slack)
dispatcher.subscribe("deploy_success", update_metrics)

dispatcher.dispatch("deploy_success", {"service": "AuthService_v2"})
```
