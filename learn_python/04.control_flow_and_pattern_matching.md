# 4. Control Flow & Structural Pattern Matching

### 4.1 Conditionals: if, elif, else, and Ternary Operator

```python
temperature = 28

if temperature > 35:
    state = "Hot"
elif temperature >= 20:
    state = "Pleasant"
else:
    state = "Cold"

# Ternary Expression: <true_val> if <condition> else <false_val>
action = "Wear Coat" if temperature < 15 else "Wear T-Shirt"
print(f"Weather: {state}, Action: {action}")
```

### 4.2 Loops: while, for, enumerate, zip, and reversed

```python
languages = ["Python", "Rust", "Go"]

# enumerate()
for idx, lang in enumerate(languages, start=1):
    print(f"{idx}: {lang}")

# zip() with strict=True (Python 3.10+)
years = [1991, 2010, 2009]
for lang, yr in zip(languages, years, strict=True):
    print(f"{lang} was released in {yr}")

# reversed()
for lang in reversed(languages):
    print("Reversed:", lang)
```

### 4.3 Loop Control: break, continue, pass, and loop-else

The **`else` clause on loops** executes **ONLY** if the loop terminates normally without hitting `break`.

```python
target = 13
numbers = [2, 4, 6, 8, 10]

for num in numbers:
    if num == target:
        print("Found target!")
        break
else:
    print("Target not found in collection.")  # Runs because no break triggered
```

### 4.4 Structural Pattern Matching (match-case)
Python 3.10+ `match-case` structural pattern matching:

```python
def handle_request(req):
    match req:
        # Literal matching
        case {"status": 200, "data": payload}:
            return f"Success with payload: {payload}"
        
        # Sequence matching with guard condition
        case ["POST", path, body] if len(body) > 0:
            return f"Handling POST to {path} with body: {body}"
        
        # Multiple pattern OR
        case 401 | 403:
            return "Unauthorized or Forbidden"
        
        # Catch-all
        case _:
            return "Unknown request format"

print(handle_request({"status": 200, "data": "User Profile"}))
print(handle_request(["POST", "/api/v1/users", "user_data"]))
print(handle_request(401))
```

### Example 1: Microservice Event Router with Structural Pattern Matching
Route heterogeneous JSON microservice payload events to specific handler methods using Python 3.10+ `match-case`.

```python
def route_event(event: dict) -> str:
    match event:
        # Match user creation with positive user_id
        case {"type": "USER_CREATED", "payload": {"id": int(uid), "email": str(email)}} if uid > 0:
            return f"Creating user profile for {email} (UID: {uid})"
        
        # Match order placed with total value
        case {"type": "ORDER_PLACED", "payload": {"order_id": str(oid), "total": float(amt)}}:
            return f"Processing order {oid} for ${amt:.2f}"
        
        # Match failure alerts with error codes
        case {"type": "SYSTEM_ALERT", "level": "CRITICAL" | "FATAL", "error_code": code}:
            return f"PAGING ON-CALL ENGINEER! Error Code: {code}"
        
        # Fallback wildcard
        case _:
            return "Discarding unrecognized event schema."

print(route_event({"type": "USER_CREATED", "payload": {"id": 42, "email": "alex@dev.internal"}}))
print(route_event({"type": "SYSTEM_ALERT", "level": "CRITICAL", "error_code": 503}))
print(route_event({"type": "UNKNOWN"}))
```

### Example 2: Network Client Exponential Backoff with Loop-`else`
Implement a reliable retry policy with exponential delay. The loop `else` block triggers only when all retry attempts fail.

```python
import time
import random

def simulated_unstable_api_call():
    # Fails 80% of the time
    if random.random() < 0.8:
        raise ConnectionError("Network reset by peer")
    return {"status": "SUCCESS", "data": "Cluster Metrics"}

def fetch_with_backoff(max_retries: int = 3) -> dict:
    delay = 0.05
    for attempt in range(1, max_retries + 1):
        try:
            print(f"[ATTEMPT {attempt}] Contacting remote API...")
            return simulated_unstable_api_call()
        except ConnectionError as err:
            print(f"Attempt {attempt} failed: {err}. Retrying in {delay:.2f}s...")
            time.sleep(delay)
            delay *= 2  # Exponential backoff
    else:
        # Only reached if loop terminates without hitting the return!
        raise TimeoutError(f"Remote API completely unreachable after {max_retries} attempts.")

try:
    data = fetch_with_backoff(max_retries=3)
    print("Fetched successfully:", data)
except TimeoutError as te:
    print("[FINAL ERROR]", te)
```

### Example 3: Chunked Batch Processor with `zip` and `enumerate`
Process large lists into fixed-sized batch chunks for bulk database insertions.

```python
def chunked_iterable(iterable, chunk_size: int):
    iterator = iter(iterable)
    while batch := list(zip(*[iterator] * chunk_size)):
        yield batch[0] if chunk_size == 1 else batch

data_stream = list(range(1, 13))
batch_size = 4

for batch_idx, batch in enumerate(zip(*[iter(data_stream)] * batch_size), start=1):
    print(f"Batch #{batch_idx}: Executing bulk DB insert of items {batch}")
```
