# 15. Essential Standard Library Modules & CLI Tools

### 15.1 Date & Time Manipulation (`datetime`, `zoneinfo`)

```python
from datetime import datetime, timezone, timedelta
from zoneinfo import ZoneInfo

# UTC aware timestamps (Always store and transmit timestamps in UTC!)
utc_now = datetime.now(timezone.utc)
print("UTC time:", utc_now.isoformat())

# Time arithmetic with timedelta
ten_days_later = utc_now + timedelta(days=10, hours=4)
print("Expiry date:", ten_days_later)

# Formatting and Parsing (strptime / strftime)
formatted = utc_now.strftime("%Y-%m-%d %H:%M:%S %Z")
print("Formatted date:", formatted)

parsed = datetime.strptime("2026-09-14 15:30:00", "%Y-%m-%d %H:%M:%S")
print("Parsed datetime:", parsed)

# Timezone conversions with zoneinfo (Python 3.9+)
try:
    ny_tz = ZoneInfo("America/New_York")
    ny_time = utc_now.astimezone(ny_tz)
    print("New York local time:", ny_time)
except Exception:
    pass
```

### 15.2 Regular Expressions with `re`

```python
import re

log_entry = "2026-09-14 12:45:00 [ERROR] user_id=4821 - Database connection timeout"

# Compiled regex with named capture groups
log_pattern = re.compile(
    r"(?P<date>\d{4}-\d{2}-\d{2})\s+"
    r"(?P<time>\d{2}:\d{2}:\d{2})\s+"
    r"\[(?P<level>[A-Z]+)\]\s+"
    r"user_id=(?P<uid>\d+)\s+-\s+"
    r"(?P<msg>.+)"
)

match = log_pattern.match(log_entry)
if match:
    data = match.groupdict()
    print("Extracted Log Details:")
    for k, v in data.items():
        print(f"  {k}: {v}")

# Substitution (cleaning text)
sanitized = re.sub(r"\d+", "XXX", "User 123 transferred 500 dollars")
print("Sanitized text:", sanitized)
```

### 15.6 Robust CLI Argument Parsing with `argparse`

`argparse` is Python's standard command-line parsing engine, supporting flags, types, help generation, and nested subcommands (like `git`).

```python
import argparse
import sys

def build_cli_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="cloudctl",
        description="Enterprise Cloud Deployment CLI Tool",
        epilog="Run 'cloudctl <subcommand> --help' for details on specific commands."
    )
    parser.add_argument("-v", "--verbose", action="store_true", help="Enable verbose diagnostic logs")
    parser.add_argument("--retries", type=int, default=3, help="Max retry attempts (default: 3)")

    # Subcommands: e.g. cloudctl deploy, cloudctl status
    subparsers = parser.add_subparsers(dest="command", required=True, help="Available subcommands")

    # Subcommand: deploy
    deploy_parser = subparsers.add_parser("deploy", help="Deploy service to cluster")
    deploy_parser.add_argument("env", choices=["staging", "production"], help="Target cloud environment")
    deploy_parser.add_argument("--image", required=True, help="Container image tag")

    # Subcommand: status
    status_parser = subparsers.add_parser("status", help="Inspect service status")
    status_parser.add_argument("--cluster-id", type=int, required=True, help="Target cluster ID")

    return parser

# Testing parser with sample CLI arguments
parser = build_cli_parser()
args = parser.parse_args(["--verbose", "deploy", "production", "--image", "app:v2.1.0"])
print(f"Parsed CLI: Command='{args.command}', Env='{args.env}', Image='{args.image}', Verbose={args.verbose}")
```

### 15.7 Cryptographic Hashing & Encoding: `hashlib` & `base64`

```python
import hashlib
import hmac
import base64

# 1. SHA-256 Cryptographic Hashing
raw_data = "SecurePayloadData_2026".encode("utf-8")
sha256_hash = hashlib.sha256(raw_data).hexdigest()
print("SHA-256 Hex:", sha256_hash)

# 2. Hash-based Message Authentication Code (HMAC) with Secret Key
secret_key = b"super_secret_auth_signing_key"
hmac_signature = hmac.new(secret_key, raw_data, hashlib.sha256).hexdigest()
print("HMAC Signature:", hmac_signature)

# Constant-time comparison to prevent timing attacks:
is_valid = hmac.compare_digest(hmac_signature, hmac.new(secret_key, raw_data, hashlib.sha256).hexdigest())
print("Constant-time HMAC valid:", is_valid)

# 3. Base64 Encoding and Decoding (safe transmission over text protocols)
encoded_b64 = base64.b64encode(b"Hello Binary World!").decode("ascii")
print("Base64 Encoded:", encoded_b64)
decoded_b64 = base64.b64decode(encoded_b64).decode("utf-8")
print("Base64 Decoded:", decoded_b64)
```

### 15.8 System Warnings & Deprecations: `warnings` module

```python
import warnings

def legacy_api_call():
    # Issue a deprecation warning without crashing the program
    warnings.warn(
        "legacy_api_call() is deprecated and will be removed in v3.0. Use modern_api_call() instead.",
        DeprecationWarning,
        stacklevel=2
    )
    return "Legacy data"

# Filter warnings: "error" turns warnings into exceptions, "ignore" silences them
with warnings.catch_warnings(record=True) as captured_warnings:
    warnings.simplefilter("always")  # Ensure all warnings are captured
    result = legacy_api_call()
    print("Function result:", result)
    print("Captured warning message:", captured_warnings[0].message)
```

### Example 2: Multi-Timezone International Meeting Coordinator
Coordinate meeting times across multiple timezones with daylight savings support using `zoneinfo`.

```python
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

def plan_global_meeting(year: int, month: int, day: int, hour: int, minute: int):
    # Scheduled time in London
    london_tz = ZoneInfo("Europe/London")
    meeting_london = datetime(year, month, day, hour, minute, tzinfo=london_tz)

    other_zones = [
        ("New York", ZoneInfo("America/New_York")),
        ("Tokyo", ZoneInfo("Asia/Tokyo")),
        ("San Francisco", ZoneInfo("America/Los_Angeles")),
        ("UTC Standard", timezone.utc)
    ]

    print(f"Meeting Origin (London): {meeting_london.strftime('%Y-%m-%d %H:%M %Z')}")
    print("Local Times for Participants:")
    for city, tz in other_zones:
        local_time = meeting_london.astimezone(tz)
        print(f"  {city:<15}: {local_time.strftime('%Y-%m-%d %H:%M %Z')}")

plan_global_meeting(2026, 9, 15, 14, 0)
```

### Example 3: Structured Access Log Parser with Named Regex Groups
Parse Nginx / Apache style access logs with regular expressions into typed dictionaries.

```python
import re

log_line = '192.168.1.50 - - [14/Sep/2026:12:00:00 +0000] "POST /api/v1/auth HTTP/1.1" 200 4096'

pattern = re.compile(
    r'(?P<ip>\d+\.\d+\.\d+\.\d+)\s+-\s+-\s+'
    r'\[(?P<timestamp>[^\]]+)\]\s+'
    r'"(?P<method>[A-Z]+)\s+(?P<path>\S+)\s+(?P<protocol>[^"]+)"\s+'
    r'(?P<status>\d{3})\s+'
    r'(?P<bytes>\d+)'
)

match = pattern.match(log_line)
if match:
    data = match.groupdict()
    print("Parsed Web Request:")
    for key, value in data.items():
        print(f"  {key:<12}: {value}")
```
