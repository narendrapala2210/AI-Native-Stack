### 21.1 Low-Level TCP Sockets: Building a Client and Echo Server

The `socket` module provides raw access to the BSD socket interface.

```python
import socket
import threading

def run_echo_server(host: str = "127.0.0.1", port: int = 65432):
    """Minimal single-threaded TCP Echo Server."""
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as server_socket:
        # Allow immediate port reuse after server restart
        server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        server_socket.bind((host, port))
        server_socket.listen(5)
        print(f"[TCP SERVER] Listening on {host}:{port}")

        conn, addr = server_socket.accept()
        with conn:
            print(f"[TCP SERVER] Connected by client at {addr}")
            while True:
                data = conn.recv(1024)
                if not data:
                    break
                print(f"[TCP SERVER] Received: {data.decode('utf-8')}")
                conn.sendall(data)  # Echo back to client

def run_tcp_client(host: str = "127.0.0.1", port: int = 65432):
    """TCP Client sending a test message."""
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as client_socket:
        client_socket.connect((host, port))
        client_socket.sendall(b"Hello over raw TCP socket!")
        response = client_socket.recv(1024)
        print(f"[TCP CLIENT] Server echo: {response.decode('utf-8')}")

# Run in separate threads for testing
server_thread = threading.Thread(target=run_echo_server, daemon=True)
server_thread.start()

import time
time.sleep(0.05)  # Wait for server to bind
run_tcp_client()
```

### 21.2 UDP Sockets & Non-Blocking Sockets

UDP (`SOCK_DGRAM`) is connectionless and lightweight (used in DNS, gaming, video streaming).

```python
import socket

# UDP Socket Demonstration
def udp_example():
    # Sender
    sender = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    
    # Receiver
    receiver = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    receiver.bind(("127.0.0.1", 9999))
    receiver.setblocking(False)  # Non-blocking mode!

    # Send datagram
    sender.sendto(b"Low-latency UDP packet", ("127.0.0.1", 9999))

    # In non-blocking mode, if no data is ready, it raises BlockingIOError instead of hanging
    try:
        data, addr = receiver.recvfrom(1024)
        print(f"[UDP] Received '{data.decode('utf-8')}' from {addr}")
    except BlockingIOError:
        print("[UDP] No packet currently waiting in socket buffer.")
    finally:
        sender.close()
        receiver.close()

udp_example()
```

### 21.3 Binary Packing with `struct` (Endianness & Network Protocols)

The `struct` module translates between Python values and C structs represented as Python `bytes`.

```python
import struct

# Packet format:
# '!' = Network (Big-Endian)
# 'I' = 4-byte unsigned int (Message ID)
# 'H' = 2-byte unsigned short (Payload length)
# '4s' = 4-byte char string (Magic code)
# 'f' = 4-byte float (Sensor reading)
packet_format = "!IH4sf"

msg_id = 1001
magic_code = b"PING"
reading = 98.6
payload_len = len(magic_code)

# 1. Packing data into raw bytes
packed_bytes = struct.pack(packet_format, msg_id, payload_len, magic_code, reading)
print(f"Packed bytes ({len(packed_bytes)} bytes): {packed_bytes.hex()}")

# 2. Unpacking raw bytes back into Python objects
unpacked = struct.unpack(packet_format, packed_bytes)
print(f"Unpacked: ID={unpacked[0]}, Length={unpacked[1]}, Magic={unpacked[2].decode()}, Reading={unpacked[3]:.1f}")
```

### 21.4 Memory-Efficient Primitive Arrays with `array.array`

When storing millions of homogeneous primitive numeric values, `array.array` bypasses Python object wrapper overhead, using compact contiguous C memory.

```python
import array
import sys

# 'i' represents signed 32-bit integer (4 bytes per element)
int_array = array.array("i", [10, 20, 30, 40, 50])
int_array.append(60)

# Memory comparison
py_list = [10, 20, 30, 40, 50, 60]
print(f"Size of list:  {sys.getsizeof(py_list)} bytes (+ object overhead per item)")
print(f"Size of array: {sys.getsizeof(int_array)} bytes (pure contiguous C array)")

# Direct binary file serialization without pickle:
with open("numbers.bin", "wb") as f:
    int_array.tofile(f)

# Direct binary file reading
restored_array = array.array("i")
with open("numbers.bin", "rb") as f:
    restored_array.fromfile(f, 6)
print("Restored binary array:", restored_array.tolist())

import os
os.remove("numbers.bin")
```

### Example 1: Multi-Client Threaded TCP Chat Server & Client
Build a multi-client echo socket server where each client connection is handled concurrently on a worker thread.

```python
import socket
import threading
import time

def handle_client(conn, addr):
    with conn:
        print(f"[THREAD] Handled connection from {addr}")
        while data := conn.recv(1024):
            conn.sendall(b"ECHO: " + data)

def run_test_server(port=54321):
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        s.bind(("127.0.0.1", port))
        s.listen(5)
        conn, addr = s.accept()
        handle_client(conn, addr)

# Run server in background
server_thread = threading.Thread(target=run_test_server, daemon=True)
server_thread.start()
time.sleep(0.02)

# Client connects
with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as client:
    client.connect(("127.0.0.1", 54321))
    client.sendall(b"Hello Server!")
    reply = client.recv(1024)
    print("Client received:", reply.decode())
```

### Example 2: UDP Heartbeat Broadcast Monitor
Broadcast real-time node heartbeat signals over UDP with timeout verification.

```python
import socket

def send_heartbeat(node_id: str, port=54322):
    with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as s:
        msg = f"HEARTBEAT:{node_id}".encode("utf-8")
        s.sendto(msg, ("127.0.0.1", port))

def listen_heartbeat(port=54322):
    with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as s:
        s.bind(("127.0.0.1", port))
        s.settimeout(0.5)
        try:
            data, addr = s.recvfrom(1024)
            print(f"[MONITOR] Node signal: {data.decode()} from {addr}")
        except socket.timeout:
            print("[MONITOR] No heartbeat signal detected within timeout.")

# Test
import threading
threading.Thread(target=listen_heartbeat, daemon=True).start()
time.sleep(0.01)
send_heartbeat("worker-node-01")
```

### Example 3: Binary Packet Marshaller with `struct`
Pack and unpack structured binary telemetry headers containing version, packet type, sequence ID, and payload length.

```python
import struct

# Format: ! (Big-Endian network order), B (1-byte version), B (1-byte type), I (4-byte sequence), H (2-byte len)
HEADER_FORMAT = "!BBIH"

def pack_telemetry_header(version: int, packet_type: int, seq_id: int, payload_len: int) -> bytes:
    return struct.pack(HEADER_FORMAT, version, packet_type, seq_id, payload_len)

def unpack_telemetry_header(raw_bytes: bytes) -> dict:
    version, p_type, seq, p_len = struct.unpack(HEADER_FORMAT, raw_bytes)
    return {"version": version, "type": p_type, "sequence": seq, "payload_len": p_len}

packed = pack_telemetry_header(version=1, packet_type=5, seq_id=10248, payload_len=512)
print("Binary Header (hex):", packed.hex())
print("Unpacked Header:", unpack_telemetry_header(packed))
```
